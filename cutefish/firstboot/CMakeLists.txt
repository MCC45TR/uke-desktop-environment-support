cmake_minimum_required(VERSION 3.22)
project(uke-cutefish-firstboot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(Pwquality REQUIRED IMPORTED_TARGET pwquality)
add_executable(uke-cutefish-firstboot-backend backend.cpp)
target_link_libraries(uke-cutefish-firstboot-backend PRIVATE PkgConfig::Pwquality)
add_executable(uke-cutefish-firstboot frontend.cpp)
target_link_libraries(uke-cutefish-firstboot PRIVATE Qt6::Widgets Qt6::Network)
include(GNUInstallDirs)
install(TARGETS uke-cutefish-firstboot uke-cutefish-firstboot-backend RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
enable_testing()
add_test(NAME provisioner-input-validation COMMAND uke-cutefish-firstboot-backend --self-test)
